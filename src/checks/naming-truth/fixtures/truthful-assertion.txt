export interface Order {
  id: string;
  lines: { sku: string; quantity: number }[];
  total: number;
}

/** Throws unless the order can be submitted. */
export function assertValidOrder(order: Order): void {
  if (order.lines.length === 0) throw new Error(`order ${order.id} has no lines`);
  if (order.total < 0) throw new Error(`order ${order.id} has a negative total`);
  for (const line of order.lines) {
    if (line.quantity <= 0) throw new Error(`order ${order.id}: ${line.sku} quantity must be positive`);
  }
}
